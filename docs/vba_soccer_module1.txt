Sub HSRatingCalc()
    Dim i As Long
    Dim j As Long
    i = Sheet3.Cells(7, 1)
    While Not IsEmpty(Sheet1.Cells(i, 2))
        Sheet3.Cells(3, 3) = i
        Sheet3.Cells(2, 1) = Sheet1.Cells(i, 2)
        Sheet3.Cells(2, 3) = Sheet1.Cells(i, 3)
        Sheet3.Cells(2, 4) = Sheet1.Cells(i, 5)
        Sheet3.Cells(2, 6) = Sheet1.Cells(i, 4)
        Sheet3.Cells(3, 2) = Sheet1.Cells(i, 6)
        Sheet3.Cells(4, 1) = Sheet1.Cells(i, 1)
        Call HSUpdater
        i = i + 1
    Wend
    Sheet3.Cells(7, 1) = i
    Beep
End Sub
